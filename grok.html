# Глава 1. Введение Ниже приведён простой и хорошо прокомментированный C# скрипт для Unity, который позволяет кубику-персонажу двигаться по осям X/Z с помощью WASD и прыгать по Space. Скрипт рассчитан на стандартный Rigidbody (физика) и подходит для Windows-сборки. Он использует `FixedUpdate` для стабильной физики и простые проверки ввода. Предполагается: - На объекте-кубике есть компонент **Rigidbody**. - Камера и уровень уже настроены (препятствия — обычные коллайдеры). - Скрипт вешается прямо на кубик. # Глава 2. Скрипт движения персонажа-кубика Создайте файл `CubeController.cs` и вставьте следующий код: ```csharp using UnityEngine; /// <summary> /// Простой контроллер движения кубика. /// WASD — ходьба, Space — прыжок. /// Требует Rigidbody на том же GameObject. /// </summary> [RequireComponent(typeof(Rigidbody))] public class CubeController : MonoBehaviour { [Header("Настройки движения")] [Tooltip("Скорость ходьбы по горизонтали")] public float moveSpeed = 5f; [Tooltip("Сила прыжка (импульс вверх)")] public float jumpForce = 7f; [Tooltip("Слой земли для проверки, стоим ли мы на поверхности")] public LayerMask groundLayer = ~0; // по умолчанию всё [Header("Проверка земли")] [Tooltip("Радиус сферы проверки под ногами")] public float groundCheckRadius = 0.2f; [Tooltip("Смещение точки проверки относительно центра кубика")] public Vector3 groundCheckOffset = new Vector3(0f, -0.5f, 0f); // Кэшированные компоненты private Rigidbody rb; // Текущий ввод private float inputX; private float inputZ; private bool jumpRequested; // Флаг «на земле» private bool isGrounded; void Awake() { // Получаем Rigidbody один раз rb = GetComponent<Rigidbody>(); // Замораживаем вращение, чтобы кубик не кувыркался rb.freezeRotation = true; } void Update() { // Считываем ввод каждый кадр (клавиатура Windows) // WASD или стрелки inputX = Input.GetAxisRaw("Horizontal"); // A/D или ←/→ inputZ = Input.GetAxisRaw("Vertical"); // W/S или ↑/↓ // Прыжок по Space (только в момент нажатия) if (Input.GetKeyDown(KeyCode.Space)) { jumpRequested = true; } } void FixedUpdate() { // Проверяем, стоим ли на земле CheckGrounded(); // Движение Move(); // Прыжок if (jumpRequested) { TryJump(); jumpRequested = false; // сбрасываем флаг } } /// <summary> /// Горизонтальное движение по X и Z. /// </summary> private void Move() { // Направление в мировых координатах Vector3 moveDirection = new Vector3(inputX, 0f, inputZ).normalized; // Желаемая скорость Vector3 targetVelocity = moveDirection * moveSpeed; // Сохраняем текущую вертикальную скорость (гравитация/прыжок) targetVelocity.y = rb.velocity.y; // Применяем скорость rb.velocity = targetVelocity; } /// <summary> /// Попытка прыжка, только если кубик на земле. /// </summary> private void TryJump() { if (isGrounded) { // Задаём импульс вверх, сохраняя горизонтальную скорость rb.velocity = new Vector3(rb.velocity.x, 0f, rb.velocity.z); rb.AddForce(Vector3.up * jumpForce, ForceMode.Impulse); } } /// <summary> /// Проверка касания земли с помощью OverlapSphere. /// </summary> private void CheckGrounded() { Vector3 checkPosition = transform.position + groundCheckOffset; isGrounded = Physics.CheckSphere(checkPosition, groundCheckRadius, groundLayer); } // Визуализация точки проверки земли в редакторе void OnDrawGizmosSelected() { Gizmos.color = Color.green; Vector3 checkPosition = transform.position + groundCheckOffset; Gizmos.DrawWireSphere(checkPosition, groundCheckRadius); } } ``` ## Как подключить 1. Добавьте на кубик компонент **Rigidbody**. 2. Навесьте скрипт `CubeController`. 3. При необходимости настройте `moveSpeed`, `jumpForce` и слой земли. 4. Убедитесь, что на препятствиях и полу есть коллайдеры. # Глава 3. Идея, как усложнить механику Чтобы расширить возможности персонажа, можно добавить **механику «рывка» (dash)** + **двойной прыжок**. ### Кратко идея - **Двойной прыжок**: разрешить один дополнительный прыжок в воздухе (флаг `canDoubleJump`). - **Рывок (Dash)** по Shift: короткий быстрый импульс в направлении движения (или взгляда). Во время рывка можно игнорировать часть препятствий или проходить сквозь «хрупкие» объекты. - Дополнительно: после рывка небольшая неуязвимость / изменение цвета кубика, чтобы игрок видел откат способности. - Управление откатами через таймеры (`dashCooldown`, `dashDuration`). Это сильно расширяет геймплей: можно преодолевать широкие пропасти, уворачиваться от движущихся препятствий и создавать секретные маршруты, требующие комбинации «двойной прыжок + рывок». Пример расширения (фрагмент логики, который можно добавить в тот же скрипт): - Ввести переменные `bool canDoubleJump`, `float dashForce`, `float dashCooldown`. - В `TryJump` разрешить второй прыжок, если `!isGrounded && canDoubleJump`. - По нажатию LeftShift запускать корутину рывка, которая на короткое время задаёт высокую скорость и затем возвращает управление. # Глава 4. Рекомендации и возможные улучшения - Добавьте анимацию/смену цвета при прыжке и рывке (через Material). - Используйте `CharacterController` вместо Rigidbody, если нужна более «аркадная» физика без скольжения. - Для более точного контроля земли можно заменить `CheckSphere` на `Raycast` или `CapsuleCast`. - Добавьте звуки шагов и прыжка через `AudioSource`. - Ограничьте максимальную скорость падения, чтобы кубик не «проваливался» сквозь пол на высоких скоростях. Скрипт выше полностью готов к использованию и легко расширяется. Удачи с мини-игрой!
